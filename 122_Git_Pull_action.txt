# github-action
# 每日、每周action更新之后
git add .
git commit -m "..."
git pull --rebase    # 把远程提交拉到本地，把自己的提交"重放"到它上面
git push
  
  ┌────────────────────────────────────────┬───────────────────────────────────────────────┬─────────────────────────────────────────────────────────────────────┐
  │                  情况                   │                    用什么                     │                                说明                                 │
  ├────────────────────────────────────────┼───────────────────────────────────────────────┼─────────────────────────────────────────────────────────────────────┤
  │ 本地有提交，远程也有新提交（最常见）      │ git pull --rebase + git push                  │ 线性历史、无多余 merge 提交；有冲突就解决后 git rebase --continue      │
  │                                        │                                               │ 再推                                                                │
  ├────────────────────────────────────────┼───────────────────────────────────────────────┼─────────────────────────────────────────────────────────────────────┤
  │ 本地没有改动，只想要最新远程              │ git pull                                      │ 纯 fast-forward，不会分叉，随便用                                    │
  ├────────────────────────────────────────┼───────────────────────────────────────────────┼─────────────────────────────────────────────────────────────────────┤
  │ 只想看看远程有什么、不想动本地            │ git fetch                                     │ 只更新远程引用，不改工作区                                            │
  ├────────────────────────────────────────┼───────────────────────────────────────────────┼─────────────────────────────────────────────────────────────────────┤
  │ 本地改到一半，想拉远程又不丢工作区改动     │ git pull --rebase --autostash                 │ 自动暂存未提交改动，rebase 完再还原                                   │
  ├────────────────────────────────────────┼───────────────────────────────────────────────┼─────────────────────────────────────────────────────────────────────┤
  │ 本地彻底乱了/不想要本地改动了             │ git clone 新目录 或 git reset --hard          │ 会丢掉本地工作，别轻易用                                              │
  │                                        │ origin/main                                   │                                                                     │
